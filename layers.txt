# happlayer 队列配置
# [queue] 开始一个新队列；队列内每行一层: 开始秒|循环(0/1)|视频路径
# 层顺序 = 从下往上叠（第一行在最底层）
# 队列内播完的层自动退出；所有「不循环」层播完 → 自动切下一队列；最后回到队列 1
# 单队列 + 全部循环 = 永不切场，不会黑场

# 窗口/显示设置（可整段删掉用默认值；命令行参数优先于此处）
[window]
size=3840x2160     # 窗口分辨率；跨屏拼接就写总尺寸，如双 1080p 横屏: 3840x1080
pos=0,0            # 窗口左上角相对「目标显示器」左上角的偏移
monitor=1          # 从第几个显示器开始（1 起，启动日志里有编号）
scale=100          # 内容缩放百分比（以窗口中心缩放，100=原始适配大小）
fullscreen=0       # 1 = 启动即全屏（占满目标显示器，此时 size/pos 无效）
borderless=1       # 1 = 无边框窗口（跨屏拼接必须设 1；Win11 下已强制直角）

[queue]
# 第一场：底层循环背景 + 2 秒入场的方块层（播完退出）+ 5 秒入场的绿块（播完退出）
0|1|clips/layer1_bottom.mov
2|0|clips/layer2_mid.mov
5|0|clips/layer3_top.mov

[queue]
# 第二场：方块层做底循环 + 1 秒入场的绿块（6 秒播完 → 整场结束，切回第一队列）
0|1|clips/layer2_mid.mov
1|0|clips/layer3_top.mov
